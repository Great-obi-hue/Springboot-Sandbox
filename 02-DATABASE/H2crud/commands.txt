// CREATE A CUSTOMER

curl -X POST "http://localhost:8080/api/v1/customers" ^
-H "Content-Type: application/json" ^
-d "{\"name\":\"John Doe\", \"address\":\"419 Lekki Phase 1\",\"gender\":\"Male\"}"

curl -X POST "http://localhost:8080/api/v1/customers" ^
-H "Content-Type: application/json" ^
-d "{\"name\":\"Jasper Adam\", \"address\":\"21 AGUDA\",\"gender\":\"Male\"}"

// VIEW ALL CUSTOMERS

curl -X GET http://localhost:8080/api/v1/customers

// VIEW PARTICULAR CUSTOMERS
curl -X GET http://localhost:8080/api/v1/customers/1
curl -X GET http://localhost:8080/api/v1/customers/2

// UPDATE A CUSTOMER
curl -X PUT "http://localhost:8080/api/v1/customers/2" ^
-H "Content-Type: application/json" ^
-d "{\"name\":\"Great Caleb\", \"address\":\"21 Surulere\",\"gender\":\"Male\"}"

// DELETE A CUSTOMER
curl -X DELETE http://localhost:8080/api/v1/customers/2
